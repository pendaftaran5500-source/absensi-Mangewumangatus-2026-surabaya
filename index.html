function linkParticipantToCheckin(data) { const lock = LockService.getScriptLock(); if (!lock.tryLock(10000)) { return { status: "error", message: "Server sedang sibuk, coba beberapa saat lagi." }; } try { initCategorySheets(); if (!data || !data.regId || !data.bib) { return { status: "error", message: "Data pendaftaran atau nomor BIB tidak lengkap!" }; } const ssReg = SpreadsheetApp.openById(CONFIG.SPREADSHEET_PENDAFTARAN_ID); const sheetReg = ssReg.getSheetByName(CONFIG.SHEET_NAME_PENDAFTARAN); const rowsReg = sheetReg.getDataRange().getValues(); const headers = rowsReg[0].map(h => h.toString().trim().toLowerCase()); const colId = headers.findIndex(h => h.includes("id pendaftar")); const colName = headers.findIndex(h => h.includes("nama utama") || h === "nama"); const colPairName = headers.findIndex(h => h.includes("nama pasangan") || h.includes("nama partner")); const colChildName = headers.findIndex(h => h.includes("nama anak")); const colCategory = headers.findIndex(h => h.includes("kategori")); const colDob = headers.findIndex(h => h.includes("tanggal lahir") || h.includes("tgl lahir")); let targetName = ""; let pairName = ""; let childName = ""; let targetCategory = ""; let matchedRegId = ""; let sheetDob = ""; const cleanRegId = data.regId.toString().trim().toLowerCase(); const inputDob = normalizeBirthDate_(data.dob || ""); for (let i = 1; i < rowsReg.length; i++) { const rowId = rowsReg[i][colId] ? rowsReg[i][colId].toString().trim() : ""; if (rowId.toLowerCase() === cleanRegId) { matchedRegId = rowId; targetName = rowsReg[i][colName] ? rowsReg[i][colName].toString().trim() : ""; if (colPairName !== -1 && rowsReg[i][colPairName]) { pairName = rowsReg[i][colPairName].toString().trim(); } if (colChildName !== -1 && rowsReg[i][colChildName]) { childName = rowsReg[i][colChildName].toString().trim(); } if (colCategory !== -1 && rowsReg[i][colCategory]) { targetCategory = rowsReg[i][colCategory].toString().trim(); } if (colDob !== -1 && rowsReg[i][colDob]) { sheetDob = normalizeBirthDate_(rowsReg[i][colDob]); } break; } } if (!matchedRegId || !targetName) { return { status: "error", message: "ID Pendaftar tidak ditemukan di database pendaftaran!" }; } if (!targetCategory) { return { status: "error", message: "Kategori peserta tidak ditemukan di database pendaftaran!" }; } const cleanBib = data.bib.toString().trim().toUpperCase().replace(/\s*&\s*/g, " & "); const bibTokens = cleanBib.split("&").map(b => b.trim()).filter(Boolean); const catLower = targetCategory.toLowerCase(); const isPairing = catLower.includes("pairing"); const isParenting = catLower.includes("parent") || catLower.includes("5km"); let displayName = targetName; if (isPairing && pairName) { displayName = `${targetName} & ${pairName}`; } else if (isParenting && childName) { displayName = `${targetName} & ${childName}`; } const ssCheckin = SpreadsheetApp.openById(CONFIG.SPREADSHEET_CHECKIN_ID); const sheetName = catLower.includes("500") ? "Checkin_500KM" : (catLower.includes("50") ? "Checkin_50KM" : "Checkin_5KM"); const sheetCheckin = ssCheckin.getSheetByName(sheetName); const existingRows = sheetCheckin.getDataRange().getValues(); // 1. Cek apakah ID Pendaftar sudah pernah diklaim sebelumnya for (let i = 1; i < existingRows.length; i++) { const existingBibStr = existingRows[i][0] ? existingRows[i][0].toString().trim().toUpperCase() : ""; const existingRegId = existingRows[i][2] ? existingRows[i][2].toString().trim().toLowerCase() : ""; if (existingRegId === matchedRegId.toLowerCase()) { if (!inputDob) { return { status: "need_dob", message: `ID Pendaftar ${matchedRegId} sudah pernah diklaim. Masukkan Tanggal Lahir untuk memulihkan akses absensi.` }; } if (!sheetDob || sheetDob !== inputDob) { return { status: "error", message: "Tanggal lahir tidak cocok dengan data pendaftaran. Hubungi panitia jika perlu bantuan." }; } return { status: "success", message: `Verifikasi berhasil! Memulihkan sesi BIB ${existingBibStr}...`, bib: existingBibStr, category: targetCategory }; } } // 2. Validasi format & rentang nomor BIB untuk klaim baru if (catLower.includes("500")) { for (const singleBib of bibTokens) { const num = parseInt(singleBib, 10); if (!/^\d+$/.test(singleBib) || isNaN(num) || num < 501 || num > 599) { return { status: "error", message: "ID Anda terdaftar di kategori 500KM. Nomor BIB wajib menggunakan rentang angka 501 sampai 599!" }; } } } else if (catLower.includes("50") && !catLower.includes("500")) { const regexB = /^B(0[1-9]|[1-9][0-9])$/; for (const singleBib of bibTokens) { if (!regexB.test(singleBib)) { return { status: "error", message: "ID Anda terdaftar di kategori 50KM. Nomor BIB wajib berformat B01 sampai B99!" }; } } } else if (catLower.includes("5") && !catLower.includes("50")) { const regexA = /^A(0[1-9]|[1-9][0-9])$/; for (const singleBib of bibTokens) { if (!regexA.test(singleBib)) { return { status: "error", message: "ID Anda terdaftar di kategori 5KM. Nomor BIB wajib berformat A01 sampai A99!" }; } } } // 3. Cek apakah nomor BIB sudah dipakai peserta lain for (let i = 1; i < existingRows.length; i++) { const existingBibStr = existingRows[i][0] ? existingRows[i][0].toString().trim().toUpperCase() : ""; const existingTokens = existingBibStr.split("&").map(b => b.trim()).filter(Boolean); for (const token of bibTokens) { if (existingTokens.includes(token)) { return { status: "error", message: `Nomor BIB ${token} sudah terdaftar di sistem check-in!` }; } } } if (sheetName === "Checkin_500KM") { sheetCheckin.appendRow(["'" + cleanBib, displayName, matchedRegId, "", "", "", "", "", "", "", "", "", "", "", ""]); } else { sheetCheckin.appendRow(["'" + cleanBib, displayName, matchedRegId, "", "", ""]); } return { status: "success", message: `BIB ${cleanBib} atas nama ${displayName} berhasil didaftarkan!`, category: targetCategory }; } finally { lock.releaseLock(); } }
